┌─ TROUBLESHOOTING ─────────────────────────────────────────────────────────────┐
│                                                                               │
│  Scripts Not Executable?                                                     │
│    cd batch && chmod +x *.sh                                                 │
│                                                                               │
│  Bun/NPM Not Found?                                                          │
│    curl -fsSL https://bun.sh/install | bash  # Install bun (recommended)    │
│    source ~/.bashrc                                                          │
│    # or: sudo apt install npm                # Install npm (alternative)    │
│                                                                               │
│  Port Already in Use?                                                        │
│    lsof -i :5001          # Find process                                     │
│    kill -9 <PID>          # Kill it                                          │
│                                                                               │
│  Python Not Found?                                                           │
│    source .venv/bin/activate                                                 │
│    ./batch/test_env.sh                                                       │
│                                                                               │
│  Agent Won't Start?                                                          │
│    tail -f logs/agent1.log                                                   │
│    ./batch/check_agents.sh                                                   │
│                                                                               │
